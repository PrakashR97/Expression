MERGE INTO ZIP_EXTRACTION_LOG T
USING (
    SELECT
        ? AS PROCESSING_ID,
        ? AS FILE_NAME,
        ? AS SOURCE_PATH,
        CAST(? AS TIMESTAMP) AS FILE_MODIFIED_DATE,
        ? AS FILE_SIZE,
        ? AS MONTH_FOLDER,
        ? AS DESTINATION_PATH
    FROM DUAL
) S
ON (T.FILE_NAME = S.FILE_NAME)

WHEN MATCHED THEN
    UPDATE SET
        T.PROCESSING_ID = S.PROCESSING_ID,
        T.SOURCE_PATH = S.SOURCE_PATH,
        T.FILE_MODIFIED_DATE = S.FILE_MODIFIED_DATE,
        T.FILE_SIZE = S.FILE_SIZE,
        T.MONTH_FOLDER = S.MONTH_FOLDER,
        T.DESTINATION_PATH = S.DESTINATION_PATH,
        T.STATUS = 'PROCESSING',
        T.START_TIME = SYSTIMESTAMP,
        T.END_TIME = NULL,
        T.ERROR_MESSAGE = NULL

WHEN NOT MATCHED THEN
    INSERT (
        PROCESSING_ID,
        FILE_NAME,
        SOURCE_PATH,
        FILE_MODIFIED_DATE,
        FILE_SIZE,
        MONTH_FOLDER,
        DESTINATION_PATH,
        STATUS,
        START_TIME
    )
    VALUES (
        S.PROCESSING_ID,
        S.FILE_NAME,
        S.SOURCE_PATH,
        S.FILE_MODIFIED_DATE,
        S.FILE_SIZE,
        S.MONTH_FOLDER,
        S.DESTINATION_PATH,
        'PROCESSING',
        SYSTIMESTAMP
    )


MERGE INTO CSV_LOAD_LOG T
USING (
    SELECT
        ? AS PROCESSING_ID,
        ? AS CSV_FILE_NAME,
        ? AS CSV_FILE_PATH,
        ? AS SOURCE_FOLDER_NAME,
        ? AS CSV_CATEGORY
    FROM DUAL
) S
ON (
    T.SOURCE_FOLDER_NAME = S.SOURCE_FOLDER_NAME
    AND T.CSV_FILE_NAME = S.CSV_FILE_NAME
)

WHEN MATCHED THEN
    UPDATE SET
        T.PROCESSING_ID = S.PROCESSING_ID,
        T.CSV_FILE_PATH = S.CSV_FILE_PATH,
        T.CSV_CATEGORY = S.CSV_CATEGORY,
        T.STATUS = 'PROCESSING',
        T.START_TIME = SYSTIMESTAMP,
        T.END_TIME = NULL,
        T.ROWS_LOADED = 0,
        T.ERROR_MESSAGE = NULL

WHEN NOT MATCHED THEN
    INSERT (
        PROCESSING_ID,
        CSV_FILE_NAME,
        CSV_FILE_PATH,
        SOURCE_FOLDER_NAME,
        CSV_CATEGORY,
        STATUS,
        START_TIME
    )
    VALUES (
        S.PROCESSING_ID,
        S.CSV_FILE_NAME,
        S.CSV_FILE_PATH,
        S.SOURCE_FOLDER_NAME,
        S.CSV_CATEGORY,
        'PROCESSING',
        SYSTIMESTAMP
    )
